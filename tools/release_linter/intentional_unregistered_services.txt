# SAMPLAR Release Linter — Intentional Unregistered Services Allowlist
#
# These classes inherit samplar::IService for the lifecycle interface
# (initialise/shutdown) but are NOT registered with the ServiceRegistry.
# They are instantiated directly as C++ objects by their owner and
# passed around via raw pointer or reference.
#
# This is an intentional architectural pattern for services that have
# a single owner and don't need global resolution via services.get().
# Triaged 2026-04-10 via release_linter check #5.
#
# ---------------------------------------------------------------------------

# Passed into audio::ProcessingAudioSource as raw pointer in the realtime path.
# Single owner, no need for global registry lookup.
PreSliceProcessingService

# Owned directly by UI pages (SampleBrowserPage / LibraryPage). The IService
# interface gives it a consistent lifecycle hook but it is not a shared service.
DrumLibraryService

# Single-owner analysis service used by WaveEditor + BeatGridOverlay.
# Instantiated per-editor-instance rather than as a shared singleton.
BeatAnalysisService

# Per-file analyzer created on demand by the sample browser.
# Short-lived; does not belong in the long-lived ServiceRegistry.
SampleAnalysisService

# Deprecated; moved to src/deprecated/ in the 2026-04-21 release. Delete after next release.
ChatService
